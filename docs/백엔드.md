## 1. 개요

### 목적
드론 이미지 세그멘테이션을 위한 **RESTful API 서버** 제공
- U-Net 모델을 활용한 실시간 부품 세그멘테이션
- 모델 성능 지표 및 학습 히스토리 제공
- 프론트엔드와 데이터 통신

### 주요 기능
- ✅ 이미지 업로드 및 세그멘테이션 예측
- ✅ 부품별 검출 정보 및 신뢰도 제공
- ✅ 이상 탐지 (부품 누락, 낮은 신뢰도)
- ✅ 모델 정보 및 성능 지표 API
- ✅ 학습 히스토리 데이터 제공
- ✅ CORS 지원 (프론트엔드 통신)

## 5. 모델 추론 모듈

### 파일: `models/unet_model.py`

#### 5.1 모델 로딩

```python
def load_model(model_path: str):
    """
    U-Net 모델 로드

    - VGG16-BN encoder with ImageNet weights
    - 6개 클래스 출력
    - 디바이스 자동 선택 (CUDA > MPS > CPU)
    """
    global MODEL, DEVICE

    DEVICE = get_device()  # GPU/MPS/CPU 자동 선택

    MODEL = smp.Unet(
        encoder_name="vgg16_bn",
        encoder_weights="imagenet",
        in_channels=3,
        classes=6
    )

    checkpoint = torch.load(model_path, map_location=DEVICE, weights_only=False)

    # 'model.' 접두사 제거 (학습 시 사용한 경우)
    state_dict = checkpoint.get("model_state_dict", checkpoint)
    new_state_dict = {}
    for key, value in state_dict.items():
        new_key = key[6:] if key.startswith("model.") else key
        new_state_dict[new_key] = value

    MODEL.load_state_dict(new_state_dict, strict=True)
    MODEL.to(DEVICE)
    MODEL.eval()
```

#### 5.2 디바이스 선택

```python
def get_device():
    """
    사용 가능한 최적의 디바이스 자동 선택

    우선순위: CUDA > MPS (Apple Silicon) > CPU
    """
    if torch.cuda.is_available():
        return torch.device("cuda")
    elif torch.backends.mps.is_available():
        return torch.device("mps")
    else:
        return torch.device("cpu")
```

#### 5.4 추론

```python
def predict(image: np.ndarray, alpha: float = 0.5) -> dict:
    """
    세그멘테이션 예측

    단계:
    1. 전처리 (512x512 리사이즈 + 정규화)
    2. 모델 추론 (logits → probabilities → argmax)
    3. 원본 크기로 복원
    4. 컬러 마스크 생성
    5. 오버레이 이미지 생성
    6. 클래스 분포 계산
    7. 신뢰도 점수 계산
    8. 이상 탐지
    """
    # 원본 크기 저장
    original_h, original_w = image.shape[:2]

    # 전처리
    image_tensor = preprocess_image(image).to(DEVICE)

    # 추론
    with torch.no_grad():
        output = MODEL(image_tensor)  # (1, 6, 512, 512)
```

#### 6.1 오버레이 생성

```python
def create_overlay(
    original_image: np.ndarray,
    colored_mask: np.ndarray,
    alpha: float = 0.5
) -> np.ndarray:
    """
    원본 이미지 위에 세그멘테이션 마스크 오버레이

    Args:
        original_image: 원본 RGB 이미지
        colored_mask: 컬러 마스크
        alpha: 투명도 (0.0 = 투명, 1.0 = 불투명)

    Returns:
        오버레이 이미지
    """
    mask_bool = np.any(colored_mask != 0, axis=2)  # 마스크 영역

    overlay = original_image.copy()
    overlay[mask_bool] = cv2.addWeighted(
        original_image[mask_bool],
        1 - alpha,
        colored_mask[mask_bool],
        alpha,
        0
    )

    return overlay
```

#### 6.2 컬러맵 적용

```python
def apply_colormap(mask: np.ndarray, colormap: dict) -> np.ndarray:
    """
    클래스 ID를 RGB 색상으로 변환

    Args:
        mask: 클래스 ID 마스크 (H, W)
        colormap: {class_id: (R, G, B)} 딕셔너리

    Returns:
        컬러 마스크 (H, W, 3)
    """
    h, w = mask.shape
    colored_mask = np.zeros((h, w, 3), dtype=np.uint8)

    for class_id, color in colormap.items():
        colored_mask[mask == class_id] = color

    return colored_mask
```

#### 7.1 IoU (Intersection over Union)

```python
def calculate_iou(
    pred_mask: np.ndarray,
    gt_mask: np.ndarray,
    num_classes: int = 6
) -> dict:
    """
    클래스별 IoU 계산

    IoU = (예측 ∩ 실제) / (예측 ∪ 실제)
    """
    ious = {}

    for cls in range(num_classes):
        pred_cls = (pred_mask == cls)
        gt_cls = (gt_mask == cls)

        intersection = np.logical_and(pred_cls, gt_cls).sum()
        union = np.logical_or(pred_cls, gt_cls).sum()

        iou = intersection / union if union > 0 else 0.0
        ious[cls] = float(iou)

    ious['mean'] = np.mean(list(ious.values()))
    return ious
```

#### 7.2 Dice Coefficient

```python
def calculate_dice(
    pred_mask: np.ndarray,
    gt_mask: np.ndarray,
    num_classes: int = 6
) -> dict:
    """
    Dice Coefficient = 2 * (예측 ∩ 실제) / (예측 + 실제)
    """
    dices = {}

    for cls in range(num_classes):
        pred_cls = (pred_mask == cls)
        gt_cls = (gt_mask == cls)

        intersection = np.logical_and(pred_cls, gt_cls).sum()
        dice = (2.0 * intersection) / (pred_cls.sum() + gt_cls.sum() + 1e-8)

        dices[cls] = float(dice)

    dices['mean'] = np.mean(list(dices.values()))
    return dices
```

#### 7.3 Precision / Recall / F1

```python
def calculate_precision_recall(
    pred_mask: np.ndarray,
    gt_mask: np.ndarray,
    num_classes: int = 6
) -> dict:
    """
    클래스별 Precision, Recall, F1 Score 계산

    Precision = TP / (TP + FP)
    Recall = TP / (TP + FN)
    F1 = 2 * (Precision * Recall) / (Precision + Recall)
    """
    metrics = {}

    for cls in range(num_classes):
        pred_cls = (pred_mask == cls)
        gt_cls = (gt_mask == cls)

        tp = np.logical_and(pred_cls, gt_cls).sum()
        fp = np.logical_and(pred_cls, ~gt_cls).sum()
        fn = np.logical_and(~pred_cls, gt_cls).sum()

        precision = tp / (tp + fp + 1e-8)
        recall = tp / (tp + fn + 1e-8)
        f1 = 2 * (precision * recall) / (precision + recall + 1e-8)

        metrics[cls] = {
            'precision': float(precision),
            'recall': float(recall),
            'f1_score': float(f1)
        }

    return metrics
```

#### 7.4 Confusion Matrix

```python
def calculate_confusion_matrix(
    pred_mask: np.ndarray,
    gt_mask: np.ndarray,
    num_classes: int = 6
) -> np.ndarray:
    """
    혼동 행렬 계산

    conf_matrix[i][j] = 실제 클래스 i를 클래스 j로 예측한 픽셀 수
    """
    conf_matrix = np.zeros((num_classes, num_classes), dtype=np.int64)

    for true_cls in range(num_classes):
        for pred_cls in range(num_classes):
            conf_matrix[true_cls, pred_cls] = np.logical_and(
                gt_mask == true_cls,
                pred_mask == pred_cls
            ).sum()

    return conf_matrix
```
